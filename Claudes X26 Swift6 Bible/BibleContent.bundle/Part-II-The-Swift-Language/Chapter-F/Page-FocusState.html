@FocusState

Kind: attribute-swiftui

Definition

A SwiftUI property wrapper that binds to the currently-focused form control. Write to it to programmatically move focus; read it to tell which field is focused.

Swift Example

struct LoginForm: View {
    enum Field { case username, password }
    @FocusState private var focus: Field?
    @State private var u = "", p = ""

    var body: some View {
        VStack {
            TextField("user", text: $u).focused($focus, equals: .username)
            SecureField("pw", text: $p).focused($focus, equals: .password)
            Button("Go") { focus = .username }
        }
    }
}
