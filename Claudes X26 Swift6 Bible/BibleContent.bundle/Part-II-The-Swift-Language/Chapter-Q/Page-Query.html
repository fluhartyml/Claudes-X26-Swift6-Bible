@Query

Kind: attribute-swiftdata

Definition

A SwiftData property wrapper that fetches a live, auto-updating collection of model objects from the model context. Views with `@Query` redraw automatically when matching data changes.

Swift Example

import SwiftData
import SwiftUI

@Model final class Note { var title: String = ""; init(title: String) { self.title = title } }

struct NotesList: View {
    @Query(sort: \.title) var notes: [Note]
    var body: some View {
        List(notes) { Text($0.title) }
    }
}
