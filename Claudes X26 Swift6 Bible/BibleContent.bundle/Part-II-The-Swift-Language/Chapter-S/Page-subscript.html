subscript

Kind: keyword

Definition

A special method that lets a type be indexed with `[…]` syntax. Subscripts can take multiple parameters, be read-only or read-write, and be generic.

Swift Example

struct Board {
    var cells: [[Int]]
    subscript(row: Int, col: Int) -> Int {
        get { cells[row][col] }
        set { cells[row][col] = newValue }
    }
}

var b = Board(cells: [[1,2],[3,4]])
print(b[0, 1]) // 2
b[1, 1] = 99
print(b.cells)
